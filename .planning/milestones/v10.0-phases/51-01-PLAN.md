```javascript
const { test, describe, before, after } = require('node:test');
const assert = require('node:assert');
const R = require('../src/engine/renderer.js');

describe('R.springConfigs registry', () => {
  test('has exactly 12 named configs', () => {
    assert.strictEqual(Object.keys(R.springConfigs).length, 12);
  });
  
  test('contains all required config names', () => {
    const required = ['buttonPress', 'iconPress', 'panelEnter', 'modalOverlay', 'modalButton', 'revealStagger', 'sceneEnter', 'hoverSpring', 'pressSpring', 'iconPressSpring', 'scrollReveal', 'entranceReveal'];
    for (const name of required) {
      assert.ok(name in R.springConfigs, `Missing config: ${name}`);
    }
  });
  
  test('buttonPress has correct params', () => {
    const cfg = R.springConfigs.buttonPress;
    assert.strictEqual(cfg.stiffness, 300);
    assert.strictEqual(cfg.damping, 24);
    assert.strictEqual(cfg.target, 0.97);
  });
  
  // ... similar tests for all 12 configs
  
  test('R.springConfig accessor works', () => {
    assert.deepStrictEqual(R.springConfig('buttonPress'), R.springConfigs.buttonPress);
    assert.strictEqual(R.springConfig('unknown'), undefined);
  });
  
  test('R.spring with registry configs produces working step functions', () => {
    const step = R.spring(R.springConfigs.buttonPress);
    const result = step(1, 0, 0.016);
    assert.ok(typeof result.value === 'number');
    assert.ok(typeof result.velocity === 'number');
  });
  
  test('spring step advances toward target', () => {
    const step = R.spring(R.springConfigs.buttonPress); // target 0.97
    let state = { value: 1, velocity: 0 };
    for (let i = 0; i < 100; i++) {
      state = step(state.value, state.velocity, 0.016);
    }
    assert.ok(Math.abs(state.value - 0.97) < 0.01);
  });
  
  test('registry is extensible', () => {
    const originalCount = Object.keys(R.springConfigs).length;
    R.springConfigs.customTest = { stiffness: 100, damping: 10, target: 1 };
    assert.strictEqual(Object.keys(R.springConfigs).length, originalCount + 1);
    assert.ok('customTest' in R.springConfigs);
    delete R.springConfigs.customTest; // cleanup
  });
});
```

| Threat ID | Category | Component | Disposition | Mitigation Plan |
