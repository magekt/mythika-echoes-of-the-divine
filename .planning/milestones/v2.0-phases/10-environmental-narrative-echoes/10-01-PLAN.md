```javascript
const NarrativeEchoes = {};

NarrativeEchoes.getForRegion = function(zoneId) {
  if (!NARRATIVE_ECHOES || !NARRATIVE_ECHOES[zoneId]) return [];
  const zoneData = NARRATIVE_ECHOES[zoneId];
  const flags = (G.state && G.state.flags) || {};
  const active = [];
  for (let i = 0; i < zoneData.echoes.length; i++) {
    const echo = zoneData.echoes[i];
    if (flags[echo.flagKey] === echo.branch) {
      active.push({ marker: echo.marker, label: echo.label, description: echo.description, markerColor: echo.markerColor });
    }
  }
  return active;
};

NarrativeEchoes.getActive = function() {
  if (!NARRATIVE_ECHOES) return [];
  const result = [];
  const zoneIds = Object.keys(NARRATIVE_ECHOES);
  for (let i = 0; i < zoneIds.length; i++) {
    const echoes = this.getForRegion(zoneIds[i]);
    if (echoes.length > 0) {
      for (let j = 0; j < echoes.length; j++) {
        result.push(Object.assign({ zoneId: zoneIds[i] }, echoes[j]));
      }
    }
  }
  return result;
};

NarrativeEchoes.isEchoed = function(zoneId) {
  return this.getForRegion(zoneId).length > 0;
};
```
