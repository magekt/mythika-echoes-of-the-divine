Evolution assist (exact):
```javascript
function requiredLevelFor(beastId, stageIdx) {
  const base = (BEAST_EVOLUTIONS[beastId] && BEAST_EVOLUTIONS[beastId][stageIdx])
    ? BEAST_EVOLUTIONS[beastId][stageIdx].level : Infinity;
  try {
    if (typeof BeastBond !== 'undefined' && BeastBond && typeof BeastBond.get === 'function'
        && BeastBond.get(beastId).heart >= 3) return Math.max(1, Math.floor(base / 2));
  } catch (e) {}
  return base;
}
```
