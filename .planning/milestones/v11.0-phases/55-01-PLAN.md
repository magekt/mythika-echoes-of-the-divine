# Phase 55 Plan 01: Element Burst Particles

## Goal

Add a lightweight pre-allocated element-colored particle burst on combat impact that reads as a distinct visual feedback layer over the damage floater. The burst is element-aware (fire=warm, water=cool, wind=light, earth=green) and crit-aware (larger, brighter). All motion is suppressed under `R.reducedMotion()` - particles appear and fade without traveling.

## Context

**Phase 54 (completed)** upgraded `R.damageNumber` to accept `opts = {element, crit}` and added spring-driven floaters. The current `src/systems/combatFx.js` already has an internal `spawnBurst` with element-keyed particles, but it allocates per-burst particle arrays on the heap. This plan migrates that behavior into a pre-allocated circular buffer in `renderer.js` (consistent with the project's zero-per-frame-allocation constraint from CONCERNS.md) and exposes a clean `R.elementBurst` API.

**Element keys in the codebase:** `fire`, `ice`, `lightning`, `dark`, `holy` (from `combatFx.js` switch). Heroes use `elementalDmgPct` bonus; skills carry optional `.element` field; `combat.js:556` resolves `const el = (skill && skill.element) || attacker.element || 'dark'`. No dedicated element enum exists in `src/data/` - elements are free-form string keys consumed by `combatFx.js` and `combat.js`.

**Color tokens:** `R.colors` has no per-element tokens yet. `combatFx.js` uses hardcoded hex strings (`#ff6020`, `#a0e0ff`, `#8040ff`, `#60ff80`, `#e8e0d0`). Phase 55 adds `R.colors.elements` map to `renderer.js` as a Phase 52-continuation of the semantic-token pattern (additive, non-breaking).

### 2. Add R.elementBurst circular buffer + spawn/update/render to renderer.js

**Files:**
- Modify: `src/engine/renderer.js`

Add after `R.updateEffects` / `R.renderEffects` (the damage-number effect block):

```js
// Phase 55: element burst particles — pre-allocated circular buffer.
// Each slot is reused in place; no object allocation per spawn.
R.elementBurstSlots = new Array(60);
R.elementBurstCount = 0;
R.elementBurstWriteIdx = 0;

R.elementBurst = function(ctx, x, y, element, crit) {
  if (G.state.reduceMotion) return;
  const el = R.colors.elements[element] || R.colors.elements.default;
  const count = crit ? 10 : 6;
  for (let i = 0; i < count; i++) {
    const slot = R.elementBurstSlots[R.elementBurstWriteIdx];
    if (slot) {
      // Reuse existing slot
      slot.x = x; slot.y = y;
      slot.vx = Math.cos((i / count) * Math.PI * 2 + Math.random() * 0.4) * (crit ? 60 : 40);
      slot.vy = Math.sin((i / count) * Math.PI * 2 + Math.random() * 0.4) * (crit ? 60 : 40);
      slot.life = crit ? 0.5 : 0.35;
      slot.maxLife = slot.life;
      slot.size = crit ? 4 : 2.5;
      slot.color = i % 2 ? el.secondary : el.primary;
      slot.active = true;
    } else {
      R.elementBurstSlots[R.elementBurstWriteIdx] = {
        x: x, y: y,
        vx: Math.cos((i / count) * Math.PI * 2 + Math.random() * 0.4) * (crit ? 60 : 40),
        vy: Math.sin((i / count) * Math.PI * 2 + Math.random() * 0.4) * (crit ? 60 : 40),
        life: crit ? 0.5 : 0.35,
        maxLife: crit ? 0.5 : 0.35,
        size: crit ? 4 : 2.5,
        color: i % 2 ? el.secondary : el.primary,
        active: true
      };
    }
    R.elementBurstWriteIdx = (R.elementBurstWriteIdx + 1) % 60;
    if (R.elementBurstCount < 60) R.elementBurstCount++;
  }
};

R.updateElementBursts = function(dt) {
  for (let i = 0; i < 60; i++) {
    const p = R.elementBurstSlots[i];
    if (!p || !p.active) continue;
    p.x += p.vx * dt;
    p.y += p.vy * dt;
    p.life -= dt;
    if (p.life <= 0) p.active = false;
  }
};

R.renderElementBursts = function(ctx) {
  const sx = R.shakeX, sy = R.shakeY;
  for (let i = 0; i < 60; i++) {
    const p = R.elementBurstSlots[i];
    if (!p || !p.active) continue;
    ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
    R.rect(ctx, p.x + sx - p.size/2, p.y + sy - p.size/2, p.size, p.size, p.color);
  }
  ctx.globalAlpha = 1;
};
```

The buffer reuses slots in place - no array `.push()` / `.splice()` on the hot path. `Math.random()` for angle jitter is acceptable (not a per-frame allocation).

### 3. Wire CombatFX.spawn to R.elementBurst

**Files:**
- Modify: `src/systems/combatFx.js`

Replace the internal `bursts` array and `spawnBurst` with a call to `R.elementBurst`. Keep the module's public API (`spawn`, `update`, `render`) so `combatScene.js` and `combat.js` don't break:

```js
const CombatFX = (function() {
  // Delegate to R.elementBurst - no internal particle state.
  function spawn(element, x, y, crit) {
    if (typeof R !== 'undefined' && R.elementBurst) {
      R.elementBurst(null, x, y, element, crit);
    }
  }

  function update(dt) {
    if (typeof R !== 'undefined' && R.updateElementBursts) R.updateElementBursts(dt);
  }

  function render(ctx) {
    if (typeof R !== 'undefined' && R.renderElementBursts) R.renderElementBursts(ctx);
  }

  return { spawn: spawn, update: update, render: render };
})();
```

### 5. Create tests/damage_burst.test.js

**Files:**
- Create: `tests/damage_burst.test.js`

Test structure (using the project's existing test pattern - `node --test`, plain `assert`):

```js
const { test } = require('node:test');
const assert = require('node:assert');

// Stub minimal R before loading renderer in test env
global.G = { state: { reduceMotion: false }, W: 400, H: 720, ctx: null, dt: 0.016 };

// Load the actual renderer
require('../src/engine/renderer.js');

test('R.elementBurst spawns particles into the buffer', () => {
  R.elementBurst(null, 100, 100, 'fire', false);
  let active = 0;
  for (let i = 0; i < 60; i++) {
    if (R.elementBurstSlots[i] && R.elementBurstSlots[i].active) active++;
  }
  assert.ok(active >= 6, 'Expected at least 6 active particles, got ' + active);
});

test('R.elementBurst with crit spawns more particles', () => {
  R.elementBurst(null, 100, 100, 'fire', true);
  let active = 0;
  for (let i = 0; i < 60; i++) {
    if (R.elementBurstSlots[i] && R.elementBurstSlots[i].active) active++;
  }
  assert.ok(active >= 10, 'Crit burst should spawn 10+ particles, got ' + active);
});

test('R.elementBurst uses element color tokens', () => {
  R.elementBurst(null, 100, 100, 'ice', false);
  const ice = R.colors.elements.ice;
  assert.ok(ice, 'R.colors.elements.ice must exist');
  assert.strictEqual(ice.primary, '#a0e0ff');
});

test('R.elementBurst with unknown element falls back to default', () => {
  R.elementBurst(null, 100, 100, 'unknown_element', false);
  const def = R.colors.elements.default;
  assert.ok(def, 'Default element colors must exist');
  assert.strictEqual(def.primary, '#e8a030');
});

test('R.reducedMotion() suppresses elementBurst', () => {
  G.state.reduceMotion = true;
  const before = R.elementBurstWriteIdx;
  R.elementBurst(null, 100, 100, 'fire', false);
  assert.strictEqual(R.elementBurstWriteIdx, before, 'No slot writes when reduced motion is on');
  G.state.reduceMotion = false;
});

test('R.updateElementBursts mutates in place - no allocation', () => {
  R.elementBurst(null, 100, 100, 'fire', false);
  // Capture slot references
  const slot0 = R.elementBurstSlots[0];
  const slot1 = R.elementBurstSlots[1];
  R.updateElementBursts(0.5);
  // Same object identity after update
  assert.strictEqual(R.elementBurstSlots[0], slot0);
  assert.strictEqual(R.elementBurstSlots[1], slot1);
});

test('R.renderElementBursts does not throw with empty buffer', () => {
  const fakeCtx = { globalAlpha: 1, fillStyle: '', fillRect: () => {} };
  R.renderElementBursts(fakeCtx);
  assert.ok(true);
});

test('R.elementBurst slots wrap around at capacity 60', () => {
  R.elementBurstWriteIdx = 0;
  R.elementBurstCount = 0;
  // Fill the buffer 10 times to force wraparound
  for (let i = 0; i < 10; i++) {
    R.elementBurst(null, 100, 100, 'fire', false);
  }
  // writeIdx should have wrapped at least once
  assert.ok(R.elementBurstWriteIdx < 60, 'writeIdx must stay in [0,60)');
});
```
