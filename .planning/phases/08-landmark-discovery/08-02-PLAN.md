```javascript
// Landmark indicators — row of small circles below progress bar
var discIds = Landmarks.getDiscovered(zoneId);
var allLandmarks = Landmarks.getAll(zoneId);
var lmX = bx + 12;
var lmY = by + bh - 20;
var lmGap = 16;
for (var i = 0; i < allLandmarks.length; i++) {
  var lm = allLandmarks[i];
  var isDisc = discIds.indexOf(lm.id) !== -1;
  var dotColor = isDisc ? R.colors.accent : R.colors.textDim;
  ctx.beginPath();
  ctx.arc(lmX + i * lmGap, lmY, 4, 0, Math.PI * 2);
  ctx.fillStyle = dotColor;
  ctx.fill();
}
```

```javascript
if (lm.isDisc) {
```
